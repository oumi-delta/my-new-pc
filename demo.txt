def totalFoldDepth (foldAt : Nat → Nat) : Nat → Nat
  | 0 => 0
  | k + 1 => totalFoldDepth foldAt k + foldAt k

def accumulatedAddition (foldAt : Nat → Nat) : Nat → Nat
  | 0 => 0
  | k + 1 =>
      3 * accumulatedAddition foldAt k
        + 2 ^ totalFoldDepth foldAt k

structure OddCycleWitness where
  length : Nat
  foldAt : Nat → Nat
  start : Nat
  length_positive : length > 0
  start_positive : start > 0
  fold_dominates :
    3 ^ length < 2 ^ totalFoldDepth foldAt length
  cycle_equation :
    (2 ^ totalFoldDepth foldAt length - 3 ^ length) * start =
      accumulatedAddition foldAt length

def knownFoldAt : Nat → Nat
  | 0 => 2
  | 1 => 2
  | _ => 0

theorem known_loop_total_depth :
    totalFoldDepth knownFoldAt 2 = 4 := by
  rfl

theorem known_loop_accumulated_addition :
    accumulatedAddition knownFoldAt 2 = 7 := by
  rfl

theorem known_loop_denominator :
    2 ^ totalFoldDepth knownFoldAt 2 - 3 ^ 2 = 7 := by
  rfl

theorem known_loop_cycle_equation :
    (2 ^ totalFoldDepth knownFoldAt 2 - 3 ^ 2) * 1 =
      accumulatedAddition knownFoldAt 2 := by
  rfl

def known_loop_witness : OddCycleWitness :=
  {
    length := 2
    foldAt := knownFoldAt
    start := 1
    length_positive := by decide
    start_positive := by decide
    fold_dominates := by decide
    cycle_equation := by rfl
  }

def nontrivialCycle (w : OddCycleWitness) : Prop :=
  w.start > 1

theorem known_loop_start_is_one :
    known_loop_witness.start = 1 := by
  rfl

theorem known_loop_is_not_nontrivial :
    ¬ nontrivialCycle known_loop_witness := by
  unfold nontrivialCycle
  decide

def cycleDenominator (w : OddCycleWitness) : Nat :=
  2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length

def cycleNumerator (w : OddCycleWitness) : Nat :=
  accumulatedAddition w.foldAt w.length

theorem nontrivial_cycle_implies_numerator_ge_two_denominator
    (w : OddCycleWitness)
    (h : nontrivialCycle w) :
    cycleDenominator w * 2 ≤ cycleNumerator w := by
  unfold nontrivialCycle at h
  unfold cycleDenominator
  unfold cycleNumerator
  have h_start_ge_two : 2 ≤ w.start := Nat.succ_le_of_lt h
  have h_mul :
      (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) * 2
        ≤
      (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) * w.start :=
    Nat.mul_le_mul_left
      (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length)
      h_start_ge_two
  rw [w.cycle_equation] at h_mul
  exact h_mul

theorem numerator_lt_two_denominator_implies_not_nontrivial
    (w : OddCycleWitness)
    (h_lt : cycleNumerator w < cycleDenominator w * 2) :
    ¬ nontrivialCycle w := by
  intro h_nontrivial
  have h_ge :
      cycleDenominator w * 2 ≤ cycleNumerator w :=
    nontrivial_cycle_implies_numerator_ge_two_denominator w h_nontrivial
  exact Nat.not_lt_of_ge h_ge h_lt

theorem numerator_lt_two_denominator_implies_start_eq_one
    (w : OddCycleWitness)
    (h_lt : cycleNumerator w < cycleDenominator w * 2) :
    w.start = 1 := by
  have h_not_nontrivial :
      ¬ nontrivialCycle w :=
    numerator_lt_two_denominator_implies_not_nontrivial w h_lt
  unfold nontrivialCycle at h_not_nontrivial
  have h_start_le_one : w.start ≤ 1 :=
    Nat.le_of_not_gt h_not_nontrivial
  have h_start_ge_one : 1 ≤ w.start :=
    w.start_positive
  exact Nat.le_antisymm h_start_le_one h_start_ge_one

theorem start_eq_one_implies_denominator_eq_numerator
    (w : OddCycleWitness)
    (h_start : w.start = 1) :
    cycleDenominator w = cycleNumerator w := by
  unfold cycleDenominator
  unfold cycleNumerator
  have h_cycle :
      (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) * w.start =
        accumulatedAddition w.foldAt w.length :=
    w.cycle_equation
  rw [h_start] at h_cycle
  simpa using h_cycle

def trivialCandidate (w : OddCycleWitness) : Prop :=

theorem cycle_equation_rearranged
    (w : OddCycleWitness) :
    cycleNumerator w + w.start * 3 ^ w.length =
      w.start * 2 ^ totalFoldDepth w.foldAt w.length := by
  unfold cycleNumerator
  have h :
      (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) * w.start =
        accumulatedAddition w.foldAt w.length :=
    w.cycle_equation
  rw [← h]
  have h_dom :
      3 ^ w.length ≤ 2 ^ totalFoldDepth w.foldAt w.length :=
    Nat.le_of_lt w.fold_dominates
  have h_sub_add :
      (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) + 3 ^ w.length =
        2 ^ totalFoldDepth w.foldAt w.length :=
    Nat.sub_add_cancel h_dom
  calc
    ((2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) * w.start)
        + w.start * 3 ^ w.length
        =
      w.start * (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length)
        + w.start * 3 ^ w.length := by
          rw [Nat.mul_comm (2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) w.start]
    _ =
      w.start *
        ((2 ^ totalFoldDepth w.foldAt w.length - 3 ^ w.length) + 3 ^ w.length) := by
          rw [← Nat.mul_add]
    _ =
      w.start * 2 ^ totalFoldDepth w.foldAt w.length := by
          rw [h_sub_add]

theorem power_two_divides_rearranged_sum
    (w : OddCycleWitness) :
    2 ^ totalFoldDepth w.foldAt w.length ∣
      cycleNumerator w + w.start * 3 ^ w.length := by
  exact Exists.intro w.start (by
    rw [cycle_equation_rearranged]
    exact Nat.mul_comm w.start
      (2 ^ totalFoldDepth w.foldAt w.length)
  )

def rearrangedSum (w : OddCycleWitness) : Nat :=
  cycleNumerator w + w.start * 3 ^ w.length

def powerTwoFactor (w : OddCycleWitness) : Nat :=
  2 ^ totalFoldDepth w.foldAt w.length

theorem rearranged_sum_has_start_as_power_two_quotient
    (w : OddCycleWitness) :
    rearrangedSum w = powerTwoFactor w * w.start := by
  unfold rearrangedSum
  unfold powerTwoFactor
  rw [cycle_equation_rearranged]
  exact Nat.mul_comm w.start
    (2 ^ totalFoldDepth w.foldAt w.length)

def oddStart (w : OddCycleWitness) : Prop :=
  w.start % 2 = 1

theorem known_loop_has_odd_start :
    oddStart known_loop_witness := by
  unfold oddStart
  rfl

theorem odd_start_not_divisible_by_two
    (w : OddCycleWitness)
    (h_odd : oddStart w) :
    ¬ 2 ∣ w.start := by
  intro h_div
  cases h_div with
  | intro k hk =>
      unfold oddStart at h_odd
      rw [hk] at h_odd
      have h_mod : (2 * k) % 2 = 0 := by
        exact Nat.mul_mod_right 2 k
      rw [h_mod] at h_odd
      contradiction

def evenStart (w : OddCycleWitness) : Prop :=

    ¬ evenStart w := by

theorem even_start_rearranged_sum_has_double_quotient
    (w : OddCycleWitness)
    (h_even : evenStart w) :
    ∃ k : Nat,
      rearrangedSum w = powerTwoFactor w * (2 * k) := by
  unfold evenStart at h_even
  cases h_even with
  | intro k hk =>
      exact Exists.intro k (by
        rw [rearranged_sum_has_start_as_power_two_quotient]
        rw [hk])

theorem even_start_power_two_factor_times_two_divides_rearranged_sum
    (w : OddCycleWitness)
    (h_even : evenStart w) :
    powerTwoFactor w * 2 ∣ rearrangedSum w := by
  have h :
      ∃ k : Nat,
        rearrangedSum w = powerTwoFactor w * (2 * k) :=
    even_start_rearranged_sum_has_double_quotient w h_even
  cases h with
  | intro k hk =>
      exact Exists.intro k (by
        rw [hk]
        rw [Nat.mul_assoc])

theorem even_start_two_divides_rearranged_sum
    (w : OddCycleWitness)
    (h_even : evenStart w) :
    2 ∣ rearrangedSum w := by
  have h :
      ∃ k : Nat,
        rearrangedSum w = powerTwoFactor w * (2 * k) :=
    even_start_rearranged_sum_has_double_quotient w h_even
  cases h with
  | intro k hk =>
      exact Exists.intro (powerTwoFactor w * k) (by
        rw [hk]
        calc
          powerTwoFactor w * (2 * k)
              = (powerTwoFactor w * 2) * k := by
                  rw [Nat.mul_assoc]
          _ = (2 * powerTwoFactor w) * k := by
                  rw [Nat.mul_comm (powerTwoFactor w) 2]
          _ = 2 * (powerTwoFactor w * k) := by
                  rw [Nat.mul_assoc]
      )

axiom start_parity_exhaustive :
  ∀ w : OddCycleWitness, oddStart w ∨ evenStart w

theorem nontrivial_cycle_is_odd_or_even_class
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    nontrivialOddStart w ∨ nontrivialEvenStart w := by
  have h_parity : oddStart w ∨ evenStart w :=
    start_parity_exhaustive w
  cases h_parity with
  | inl h_odd =>
      exact Or.inl ⟨h_nontrivial, h_odd⟩
  | inr h_even =>
      exact Or.inr ⟨h_nontrivial, h_even⟩

theorem nontrivial_cycle_forces_odd_or_even_burden
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    (∃ q : Nat,
      rearrangedSum w = powerTwoFactor w * q ∧
      q > 1 ∧
      q % 2 = 1) ∨
    (powerTwoFactor w * 2 ∣ rearrangedSum w) := by
  have h_class :
      nontrivialOddStart w ∨ nontrivialEvenStart w :=
    nontrivial_cycle_is_odd_or_even_class w h_nontrivial
  cases h_class with
  | inl h_odd_class =>
      exact Or.inl
        (nontrivial_odd_start_has_odd_quotient_gt_one_from_class w h_odd_class)
  | inr h_even_class =>
      exact Or.inr
        (nontrivial_even_start_has_double_power_two_divisibility_from_class w h_even_class)

theorem nontrivial_cycle_full_burden_summary
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    ¬ trivialCandidate w ∧
    (
      (∃ q : Nat,
        rearrangedSum w = powerTwoFactor w * q ∧
        q > 1 ∧
        q % 2 = 1) ∨
      (powerTwoFactor w * 2 ∣ rearrangedSum w)
    ) := by
  constructor
  · exact nontrivial_cycle_not_trivial_candidate w h_nontrivial
  · exact nontrivial_cycle_forces_odd_or_even_burden w h_nontrivial

def oddBurden (w : OddCycleWitness) : Prop :=
  ∃ q : Nat,
    rearrangedSum w = powerTwoFactor w * q ∧
    q > 1 ∧
    q % 2 = 1

def evenBurden (w : OddCycleWitness) : Prop :=
  powerTwoFactor w * 2 ∣ rearrangedSum w

theorem nontrivial_cycle_has_burden_class
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    ¬ trivialCandidate w ∧
      (oddBurden w ∨ evenBurden w) := by
  have h_summary :
      ¬ trivialCandidate w ∧
      (
        (∃ q : Nat,
          rearrangedSum w = powerTwoFactor w * q ∧
          q > 1 ∧
          q % 2 = 1) ∨
        (powerTwoFactor w * 2 ∣ rearrangedSum w)
      ) :=
    nontrivial_cycle_full_burden_summary w h_nontrivial
  constructor
  · exact h_summary.left
  · cases h_summary.right with
    | inl h_odd =>
        exact Or.inl h_odd
    | inr h_even =>
        exact Or.inr h_even

def nontrivialBurdenClass (w : OddCycleWitness) : Prop :=
  ¬ trivialCandidate w ∧ (oddBurden w ∨ evenBurden w)

theorem nontrivial_cycle_enters_burden_class
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    nontrivialBurdenClass w := by
  unfold nontrivialBurdenClass
  exact nontrivial_cycle_has_burden_class w h_nontrivial

theorem burden_class_if_not_odd_then_even
    (w : OddCycleWitness)
    (h : nontrivialBurdenClass w)
    (h_not_odd : ¬ oddBurden w) :
    evenBurden w := by
  unfold nontrivialBurdenClass at h
  cases h.right with
  | inl h_odd =>
      exact False.elim (h_not_odd h_odd)
  | inr h_even =>
      exact h_even

theorem burden_class_if_not_even_then_odd
    (w : OddCycleWitness)
    (h : nontrivialBurdenClass w)
    (h_not_even : ¬ evenBurden w) :
    oddBurden w := by
  unfold nontrivialBurdenClass at h
  cases h.right with
  | inl h_odd =>
      exact h_odd
  | inr h_even =>
      exact False.elim (h_not_even h_even)

theorem burden_class_cannot_escape_both
    (w : OddCycleWitness)
    (h : nontrivialBurdenClass w) :
    ¬ (¬ oddBurden w ∧ ¬ evenBurden w) := by
  intro h_escape
  unfold nontrivialBurdenClass at h
  cases h.right with
  | inl h_odd =>
      exact h_escape.left h_odd
  | inr h_even =>
      exact h_escape.right h_even

theorem eliminating_both_burdens_eliminates_burden_class
    (w : OddCycleWitness)
    (h_not_odd : ¬ oddBurden w)
    (h_not_even : ¬ evenBurden w) :
    ¬ nontrivialBurdenClass w := by
  intro h_class
  have h_no_escape :
      ¬ (¬ oddBurden w ∧ ¬ evenBurden w) :=
    burden_class_cannot_escape_both w h_class
  exact h_no_escape ⟨h_not_odd, h_not_even⟩

theorem nontrivial_cycle_impossible_if_both_burdens_eliminated
    (w : OddCycleWitness)
    (h_not_odd : ¬ oddBurden w)
    (h_not_even : ¬ evenBurden w) :
    ¬ nontrivialCycle w := by
  intro h_nontrivial
  have h_class : nontrivialBurdenClass w :=
    nontrivial_cycle_enters_burden_class w h_nontrivial
  have h_not_class : ¬ nontrivialBurdenClass w :=
    eliminating_both_burdens_eliminates_burden_class
      w h_not_odd h_not_even
  exact h_not_class h_class

def burdensEliminated (w : OddCycleWitness) : Prop :=
  ¬ oddBurden w ∧ ¬ evenBurden w

theorem burdens_eliminated_eliminates_burden_class
    (w : OddCycleWitness)
    (h : burdensEliminated w) :
    ¬ nontrivialBurdenClass w := by
  exact eliminating_both_burdens_eliminates_burden_class

theorem burdens_eliminated_eliminates_nontrivial_cycle
    (w : OddCycleWitness)
    (h : burdensEliminated w) :

theorem odd_burden_has_quotient_gt_one
    (w : OddCycleWitness)
    (h : oddBurden w) :
    ∃ q : Nat,
      q > 1 ∧
      q % 2 = 1 ∧
      rearrangedSum w = powerTwoFactor w * q := by
  unfold oddBurden at h
  cases h with
  | intro q hq =>
      exact Exists.intro q (by
        constructor
        · exact hq.right.left
        · constructor
          · exact hq.right.right
          · exact hq.left
      )

theorem even_burden_has_double_power_two_factor
    (w : OddCycleWitness)
    (h : evenBurden w) :
    ∃ q : Nat,
      rearrangedSum w = (powerTwoFactor w * 2) * q := by
  unfold evenBurden at h
  exact h

theorem odd_burden_quotient_at_least_three
    (w : OddCycleWitness)
    (h : oddBurden w) :
    ∃ q : Nat,
      rearrangedSum w = powerTwoFactor w * q ∧
      3 ≤ q ∧
      q % 2 = 1 := by
  unfold oddBurden at h
  cases h with
  | intro q hq =>
      have h_gt_one : q > 1 := hq.right.left
      have h_odd : q % 2 = 1 := hq.right.right
      have h_not_two : q ≠ 2 := by
        intro h_eq_two
        rw [h_eq_two] at h_odd
        contradiction
      have h_ge_two : 2 ≤ q := Nat.succ_le_of_lt h_gt_one
      have h_ge_three : 3 ≤ q := by
        cases h_ge_two with
        | refl =>
            exact False.elim (h_not_two rfl)
        | step h_two_le_pred =>
            exact Nat.succ_le_succ h_two_le_pred
      exact Exists.intro q (by
        constructor
        · exact hq.left
        · constructor
          · exact h_ge_three
          · exact h_odd
      )

theorem odd_burden_rearranged_sum_ge_three_power_two_factor
    (w : OddCycleWitness)
    (h : oddBurden w) :
    powerTwoFactor w * 3 ≤ rearrangedSum w := by
  have h_three :
      ∃ q : Nat,
        rearrangedSum w = powerTwoFactor w * q ∧
        3 ≤ q ∧
        q % 2 = 1 :=
    odd_burden_quotient_at_least_three w h
  cases h_three with
  | intro q hq =>
      have h_q_ge_three : 3 ≤ q := hq.right.left
      have h_mul :
          powerTwoFactor w * 3 ≤ powerTwoFactor w * q :=
        Nat.mul_le_mul_left (powerTwoFactor w) h_q_ge_three
      rw [← hq.left] at h_mul
      exact h_mul

theorem even_burden_rearranged_sum_ge_two_power_two_factor
    (w : OddCycleWitness)
    (h_even : evenStart w) :
    powerTwoFactor w * 2 ≤ rearrangedSum w := by
  unfold evenStart at h_even
  cases h_even with
  | intro k hk =>
      have h_k_pos : 1 ≤ k := by
        cases k with
        | zero =>
            have h_start_pos : w.start > 0 := w.start_positive
            rw [Nat.mul_zero] at hk
            rw [hk] at h_start_pos
            exact False.elim ((Nat.lt_irrefl 0) h_start_pos)
        | succ k' =>
            exact Nat.succ_le_succ (Nat.zero_le k')
      have h_eq :
          rearrangedSum w = powerTwoFactor w * (2 * k) := by
        rw [rearranged_sum_has_start_as_power_two_quotient]
        rw [hk]
      have h_base :
          powerTwoFactor w * 2 ≤ powerTwoFactor w * (2 * k) := by
        have h_two_le_two_mul_k : 2 ≤ 2 * k := by
          have h_mul : 2 * 1 ≤ 2 * k :=
            Nat.mul_le_mul_left 2 h_k_pos
          simpa using h_mul
        exact Nat.mul_le_mul_left (powerTwoFactor w) h_two_le_two_mul_k
      rw [← h_eq] at h_base
      exact h_base

theorem nontrivial_even_start_rearranged_sum_ge_two_power_two_factor

theorem nontrivial_odd_start_rearranged_sum_ge_three_power_two_factor

theorem nontrivial_cycle_odd_or_even_lower_bound
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    (nontrivialOddStart w ∧
      powerTwoFactor w * 3 ≤ rearrangedSum w) ∨
    (nontrivialEvenStart w ∧
      powerTwoFactor w * 2 ≤ rearrangedSum w) := by
  have h_class :
      nontrivialOddStart w ∨ nontrivialEvenStart w :=
    nontrivial_cycle_is_odd_or_even_class w h_nontrivial
  cases h_class with
  | inl h_odd =>
      exact Or.inl (by
        constructor
        · exact h_odd
        · exact nontrivial_odd_start_rearranged_sum_ge_three_power_two_factor
            w h_odd
      )
  | inr h_even =>
      exact Or.inr (by
        constructor
        · exact h_even
        · exact nontrivial_even_start_rearranged_sum_ge_two_power_two_factor
            w h_even
      )

def nontrivialLowerBoundClass (w : OddCycleWitness) : Prop :=
  (nontrivialOddStart w ∧
    powerTwoFactor w * 3 ≤ rearrangedSum w) ∨
  (nontrivialEvenStart w ∧
    powerTwoFactor w * 2 ≤ rearrangedSum w)

theorem nontrivial_cycle_enters_lower_bound_class
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    nontrivialLowerBoundClass w := by
  unfold nontrivialLowerBoundClass
  exact nontrivial_cycle_odd_or_even_lower_bound w h_nontrivial

theorem lower_bound_class_impossible_below_two_power_two_factor
    (w : OddCycleWitness)
    (h_class : nontrivialLowerBoundClass w)
    (h_lt : rearrangedSum w < powerTwoFactor w * 2) :
    False := by
  unfold nontrivialLowerBoundClass at h_class
  cases h_class with
  | inl h_odd =>
      have h_ge_three :
          powerTwoFactor w * 3 ≤ rearrangedSum w :=
        h_odd.right
      have h_two_le_three :
          powerTwoFactor w * 2 ≤ powerTwoFactor w * 3 := by
        have h_2_le_3 : 2 ≤ 3 := by decide
        exact Nat.mul_le_mul_left (powerTwoFactor w) h_2_le_3
      have h_ge_two :
          powerTwoFactor w * 2 ≤ rearrangedSum w :=
        Nat.le_trans h_two_le_three h_ge_three
      exact Nat.not_lt_of_ge h_ge_two h_lt
  | inr h_even =>
      have h_ge_two :
          powerTwoFactor w * 2 ≤ rearrangedSum w :=
        h_even.right
      exact Nat.not_lt_of_ge h_ge_two h_lt

theorem nontrivial_cycle_impossible_below_two_power_two_factor
    (w : OddCycleWitness)
    (h_lt : rearrangedSum w < powerTwoFactor w * 2) :
    ¬ nontrivialCycle w := by
  intro h_nontrivial
  have h_class : nontrivialLowerBoundClass w :=
    nontrivial_cycle_enters_lower_bound_class w h_nontrivial
  exact lower_bound_class_impossible_below_two_power_two_factor
    w h_class h_lt

theorem lower_bound_class_gives_two_power_two_factor_lower_bound
    (w : OddCycleWitness)
    (h_class : nontrivialLowerBoundClass w) :
    powerTwoFactor w * 2 ≤ rearrangedSum w := by
  unfold nontrivialLowerBoundClass at h_class
  cases h_class with
  | inl h_odd =>
      have h_ge_three :
          powerTwoFactor w * 3 ≤ rearrangedSum w :=
        h_odd.right
      have h_two_le_three :
          powerTwoFactor w * 2 ≤ powerTwoFactor w * 3 := by
        have h_2_le_3 : 2 ≤ 3 := by decide
        exact Nat.mul_le_mul_left (powerTwoFactor w) h_2_le_3
      exact Nat.le_trans h_two_le_three h_ge_three
  | inr h_even =>
      exact h_even.right

theorem nontrivial_cycle_gives_two_power_two_factor_lower_bound
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    powerTwoFactor w * 2 ≤ rearrangedSum w := by
  have h_class : nontrivialLowerBoundClass w :=
    nontrivial_cycle_enters_lower_bound_class w h_nontrivial
  exact lower_bound_class_gives_two_power_two_factor_lower_bound
    w h_class

def twoPowerTwoFactorLowerBound (w : OddCycleWitness) : Prop :=
  powerTwoFactor w * 2 ≤ rearrangedSum w

def belowTwoPowerTwoFactor (w : OddCycleWitness) : Prop :=
  rearrangedSum w < powerTwoFactor w * 2

theorem nontrivial_cycle_enters_two_power_two_factor_lower_bound
    (w : OddCycleWitness)
    (h_nontrivial : nontrivialCycle w) :
    twoPowerTwoFactorLowerBound w := by
  unfold twoPowerTwoFactorLowerBound
  exact nontrivial_cycle_gives_two_power_two_factor_lower_bound
    w h_nontrivial

theorem below_two_power_two_factor_eliminates_nontrivial_cycle
    (w : OddCycleWitness)
    (h_below : belowTwoPowerTwoFactor w) :
    ¬ nontrivialCycle w := by
  unfold belowTwoPowerTwoFactor at h_below
  exact nontrivial_cycle_impossible_below_two_power_two_factor
    w h_below

theorem two_power_lower_bound_and_below_contradiction
    (w : OddCycleWitness)
    (h_lower : twoPowerTwoFactorLowerBound w)
    (h_below : belowTwoPowerTwoFactor w) :
    False := by
  unfold twoPowerTwoFactorLowerBound at h_lower
  unfold belowTwoPowerTwoFactor at h_below
  exact Nat.not_lt_of_ge h_lower h_below
